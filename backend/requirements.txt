# Runtime dependencies. Every package directly imported by app/ is pinned here
# explicitly rather than relied on transitively, so an upstream dependency
# dropping it cannot silently break the image.

# Web framework and ASGI server
fastapi==0.139.2
uvicorn==0.51.0
# Required by FastAPI to parse multipart/form-data: file uploads and the
# OAuth2 password form used by /api/v1/auth/login.
python-multipart==0.0.32

# Models, validation and settings
pydantic==2.13.4
pydantic-settings==2.7.1
# Backs pydantic's EmailStr.
email-validator==2.2.0

# Database: async SQLAlchemy over asyncpg, migrations via Alembic.
# The [asyncio] extra pulls in greenlet, which SQLAlchemy's async engine
# requires and does not install on every platform by default.
sqlalchemy[asyncio]==2.0.36
asyncpg==0.30.0
alembic==1.14.0

# Authentication: JWT signing plus password hashing.
# bcrypt is pinned below 5.0 because passlib 1.7.4 reads bcrypt.__about__,
# which newer releases removed.
#
# python-jose pulls in ecdsa, which pip-audit flags for PYSEC-2026-1325 (Minerva
# timing attack). There is no fixed release and the ecdsa project treats
# side-channel resistance as out of scope. It only affects ECDSA signing/keygen
# and we sign with HS256, so the vulnerable path is never reached. Revisit if
# jwt_algorithm is ever switched to an ES* algorithm.
python-jose[cryptography]==3.5.0
passlib[bcrypt]==1.7.4
bcrypt==4.0.1

# Upload handling: detects the real image format from file bytes.
pillow==12.3.0

# HTTP client used by the AI service adapter.
httpx==0.27.0
